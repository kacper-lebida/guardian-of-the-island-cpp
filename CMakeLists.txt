cmake_minimum_required(VERSION 3.16)
project(guardian_of_the_island LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rpg_core src/Character.cpp src/Game.cpp src/Inventory.cpp src/Item.cpp src/MenuSystem.cpp src/SaveSystem.cpp)
target_include_directories(rpg_core PUBLIC src)
if(MSVC)
    target_compile_options(rpg_core PRIVATE /W4)
else()
    target_compile_options(rpg_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(guardian src/main.cpp)
target_link_libraries(guardian PRIVATE rpg_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(regression tests/regression.cpp)
    target_link_libraries(regression PRIVATE rpg_core)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-work")
    add_test(NAME core_regression COMMAND regression)
    set_tests_properties(core_regression PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-work")
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" "$<TARGET_FILE:guardian>")
    endif()
endif()
